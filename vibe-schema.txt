
Table Hospital {
  id String [pk, default: `uuid()`, note: 'Định danh duy nhất của bệnh viện']
  code String [not null, unique, note: 'Mã bệnh viện, duy nhất trên toàn hệ thống']
  name String [not null, note: 'Tên bệnh viện']
  address String [note: 'Địa chỉ bệnh viện']
  phone String [note: 'Số điện thoại liên hệ']
  timezone String [not null, default: 'Asia/Ho_Chi_Minh', note: 'Múi giờ sử dụng để xử lý lịch khám và thời gian']
  status HospitalStatus [not null, default: 'ACTIVE', note: 'Trạng thái hoạt động của bệnh viện']
  createdAt DateTime [not null, default: `now()`, note: 'Thời điểm tạo bệnh viện']
  updatedAt DateTime [not null, note: 'Thời điểm cập nhật gần nhất']

  Note: 'Quản lý thông tin bệnh viện. Đây là Tenant gốc, phân tách dữ liệu giữa các bệnh viện trong hệ thống Multi-tenant.'
}
